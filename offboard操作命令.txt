# PX4 + MAVROS + Offboard 起飞与常用监控命令
# 适用：Ubuntu 20.04 / ROS Noetic / PX4 / MAVROS
# 坐标约定：MAVROS local 使用 ENU，x 向前，y 向左，z 向上。
# 注意：下面的命令分为“仿真”和“实机”两种场景，不要混用串口和 SITL 命令。


# ============================================================
# 一、每个终端先加载 ROS 环境
# ============================================================

source /opt/ros/noetic/setup.bash

# 如果终端出现 (base)，建议先退出 Conda：
conda deactivate 2>/dev/null || true


# ============================================================
# 二、仿真环境：Gazebo + PX4 + MAVROS
# ============================================================

# 比赛森林地图示例。该入口会同时启动 PX4、Gazebo 和 MAVROS。
source /opt/ros/noetic/setup.bash
source "$HOME/planning_ws/devel/setup.bash"
source "$HOME/PX4_Firmware/Tools/setup_gazebo.bash" \
  "$HOME/PX4_Firmware" \
  "$HOME/PX4_Firmware/build/px4_sitl_default"
export ROS_PACKAGE_PATH="$ROS_PACKAGE_PATH:$HOME/PX4_Firmware:$HOME/PX4_Firmware/Tools/sitl_gazebo"

roslaunch px4 orienteering_competition.launch \
  x:=0 y:=-27.5 z:=9.2 R:=0 P:=0 Y:=0 \
  gui:=true paused:=false interactive:=false \
  world:="$HOME/PX4_Firmware/Tools/sitl_gazebo/worlds/orienteering_realistic.world"

# 如果只需要普通空场景，可使用 PX4 官方 SITL 入口：
# roslaunch px4 mavros_posix_sitl.launch


# ============================================================
# 三、实机：启动 MAVROS 串口连接
# ============================================================

# NX 上 MAVROS 源码工作空间的示例：
source /opt/ros/noetic/setup.bash
source "$HOME/mavros_ws/devel/setup.bash"

# 飞控连接在 /dev/ttyACM0，波特率 921600：
roslaunch mavros px4.launch \
  fcu_url:=serial:///dev/ttyACM0:921600

# 如果同时把 MAVLink 转发给 Windows QGroundControl：
# Windows IP 示例：10.160.52.250，QGC UDP 端口：14550
roslaunch mavros px4.launch \
  fcu_url:=serial:///dev/ttyACM0:921600 \
  gcs_url:=udp://@10.160.52.250:14550

# 同一个飞控串口只能由一个 MAVROS/QGC 实例占用。


# ============================================================
# 四、启动后必须先检查的状态
# ============================================================

# MAVROS 是否连接飞控：
rostopic echo -n 1 /mavros/state

# 正常应看到：
# connected: True

# 本地位姿是否发布：
rostopic echo -n 1 /mavros/local_position/pose

# 本地里程计频率：
rostopic hz /mavros/local_position/odom

# 当前飞行模式、解锁状态和连接状态：
rostopic echo /mavros/state

# 当前 PX4 状态文字，包括 failsafe 原因：
rostopic echo /mavros/statustext/recv


# ============================================================
# 五、常用传感器与飞控数据
# ============================================================

# 电池：
rostopic echo -n 1 /mavros/battery

# GPS 原始定位：
rostopic echo -n 1 /mavros/global_position/raw/fix

# GPS 状态、卫星数、精度：
rostopic echo -n 1 /mavros/gpsstatus/gps1/raw

# 融合后的全局位置：
rostopic echo -n 1 /mavros/global_position/global

# 融合后的相对高度：
rostopic echo -n 1 /mavros/global_position/rel_alt

# IMU：
rostopic echo -n 1 /mavros/imu/data

# 气压计静态压力：
rostopic echo -n 1 /mavros/imu/static_pressure

# 气压计温度：
rostopic echo -n 1 /mavros/imu/temperature_baro

# 本地速度：
rostopic echo -n 1 /mavros/local_position/velocity_local

# RC 输入：
rostopic echo -n 1 /mavros/rc/in

# 当前所有 MAVROS 话题：
rostopic list | grep '^/mavros'


# ============================================================
# 六、启动 Offboard 1 米起飞测试
# ============================================================

# 先确认 MAVROS connected=True 且 local_position/pose 有数据。
# 再打开新终端，加载 ROS 环境后运行：

source /opt/ros/noetic/setup.bash
python3 ~/offboard_takeoff_1m.py \
  _height:=1.0 \
  _hold_time:=5.0 \
  _reach_timeout:=20.0 \
  _position_tolerance:=0.15 \
  _land_after:=true

# 该脚本会：
# 1. 读取当前 local pose；
# 2. 将目标设为当前位置上方 1 m；
# 3. 以 20 Hz 连续发布 /mavros/setpoint_position/local；
# 4. 预发布 3 秒设定点；
# 5. 切换 OFFBOARD；
# 6. 解锁并上升到目标高度；
# 7. 悬停 5 秒；
# 8. 请求 AUTO.LAND。

# 只悬停、不自动降落的测试：
# python3 ~/offboard_takeoff_1m.py \
#   _height:=1.0 _hold_time:=5.0 _land_after:=false
# 该模式必须保持脚本运行；停止脚本会触发 Offboard 丢失保护。


# ============================================================
# 七、Offboard 关键诊断
# ============================================================

# 设定点是否持续发布，正常约 20 Hz：
rostopic hz /mavros/setpoint_position/local

# 最近一条设定点：
rostopic echo -n 1 /mavros/setpoint_position/local

# 如果使用 raw setpoint，则检查：
rostopic hz /mavros/setpoint_raw/local

# 检查 Offboard 相关 PX4 参数：
rosservice call /mavros/param/get "param_id: COM_OF_LOSS_T"
rosservice call /mavros/param/get "param_id: COM_OBL_ACT"

# 出现以下信息时，通常是设定点停止或频率过低：
# No offboard signal
# Failsafe activated
# Autopilot disengaged

# Offboard 模式下不要关闭持续发布设定点的节点。


# ============================================================
# 八、手动切换模式与紧急降落
# ============================================================

# 手动请求 OFFBOARD（必须已有持续设定点）：
rosservice call /mavros/set_mode \
  "{base_mode: 0, custom_mode: 'OFFBOARD'}"

# 手动解锁：
rosservice call /mavros/cmd/arming "{value: true}"

# 紧急请求自动降落：
rosservice call /mavros/set_mode \
  "{base_mode: 0, custom_mode: 'AUTO.LAND'}"

# 降落后确认已上锁：
rostopic echo -n 1 /mavros/state
# armed: False


# ============================================================
# 九、停止流程
# ============================================================

# 1. 先确认 AUTO.LAND 已完成、armed=False；
# 2. 再在 Offboard 脚本终端按 Ctrl+C；
# 3. 再停止 MAVROS/Gazebo/其他节点。
# 不要在飞机仍处于 OFFBOARD 且已解锁时直接关闭设定点发布节点。
